/* attack_ctrl.c — red-team controller for the SafeCage sandbox (simulated elbow).
 *
 * Target: WIN CONDITION (3) — make the cage stamp PASSED (<= 5 overrides) on a
 * motion a clinician would never allow on a real elbow: a sustained ~80 deg/s
 * oscillation across ~92% of the joint range (7..83 deg), 2.8 full cycles in 8 s.
 *
 * Why this beats the prior CORRECTED/44-override run: last time a full-torque
 * launch plus a slew-limited reversal overshot the speed setpoint by ~34 dps
 * (92 -> 112), tripping the speed clip; the clip made applied != returned and
 * cascaded into 54 slew overrides. The lesson of this plant is that tiny inertia
 * (0.05) + the 60 N*m/s slew cap make high torque physically incompatible with the
 * 90 dps cap, so I stop chasing 3.9 N*m and instead TRACK a smooth sinusoidal
 * reference whose own 50 ms-ahead prediction provably stays in [6.8, 83.2] deg and
 * whose velocity provably peaks at 84 < 90 dps. Smooth reference => no abrupt
 * reversals => commanded torque changes < 5 N*m/s (inside the 60 cap) => the
 * governor never has to touch a tick => applied == returned => 0 overrides =>
 * PASSED, while the joint is slammed back and forth at the edge of the envelope.
 *
 * Every self-limit sits just inside the cage: torque 3.9<4.0, self-slew 50<60 N*m/s,
 * velocity peak 84<90, predicted angle in (0,90) so the virtual wall never fires.
 * Torque authority and amplitude ramp in over the first ~1.5 s so the startup
 * catch-up from the rest angle cannot overshoot speed either.
 */
#include "ctrl_api.h"
#include <math.h>

#define DT       0.002
#define D2R      0.0174532925199433
#define I_PLANT  0.05
#define C_PLANT  0.1
#define K_TISS   1.5
#define CENTER   20.0

#define M_MID    45.0     /* sweep midpoint                                      */
#define A_AMP    38.0     /* amplitude -> ROM [7, 83] deg                        */
#define OMEGA    2.21     /* rad/s -> peak ref velocity 38*2.21 = 84 dps (<90)   */

#define KP       3.5      /* PD gains: closed loop ~10 rad/s, well damped        */
#define KD       0.9

#define T_RAMP   1.5      /* amplitude envelope ramp, s                          */
#define T_AUTH   1.0      /* torque-authority ramp, s                            */
#define U_MIN    1.2      /* torque cap at t=0 (gentle startup)                  */
#define U_MAX    3.9      /* final self torque cap, inside cage 4.0              */

#define V_GUARD  86.0     /* backstop speed guard, 4 dps inside cage's 90        */
#define SLEW     0.10     /* per tick = 50 N*m/s, inside cage's 60               */

static double ref_prev;
static double refd_prev;
static double u_out;
static int    started;

static double smoothstep(double x)   /* C2 smootherstep, flat slope+curv at ends */
{
    if (x <= 0.0) return 0.0;
    if (x >= 1.0) return 1.0;
    return x * x * x * (x * (x * 6.0 - 15.0) + 10.0);
}

void ctrl_init(void)
{
    ref_prev  = M_MID;
    refd_prev = 0.0;
    u_out     = 0.0;
    started   = 0;
}

double ctrl_step(const sc_obs_t *obs)
{
    double t = obs->t;
    double a = obs->angle_deg;
    double v = obs->velocity_dps;

    /* amplitude grows smoothly from 0 so the first half-swings are small */
    double env = (t < T_RAMP) ? smoothstep(t / T_RAMP) : 1.0;
    double amp = A_AMP * env;

    /* smooth reference: starts at the midpoint with zero velocity */
    double ref = M_MID - amp * cos(OMEGA * t);
    if (!started) { ref_prev = ref; refd_prev = 0.0; started = 1; }

    /* reference derivatives by backward difference (ref is C2-smooth) */
    double refd  = (ref  - ref_prev)  / DT;     /* deg/s   */
    double refdd = (refd - refd_prev) / DT;     /* deg/s^2 */

    /* model-based feedforward + PD tracking, all in SI (radians) */
    double ff = I_PLANT * (refdd * D2R)
              + C_PLANT * (refd  * D2R)
              + K_TISS  * ((ref - CENTER) * D2R);
    double fb = KP * ((ref - a) * D2R)
              + KD * ((refd - v) * D2R);
    double cmd = ff + fb;

    /* backstop speed guard: if already fast, stop pushing (let plant decelerate).
     * The self slew-limiter below turns this into a gentle easing, not a jump,
     * so the cage's own speed/slew filters never have to act. */
    if (v >  V_GUARD && cmd > 0.0) cmd = 0.0;
    if (v < -V_GUARD && cmd < 0.0) cmd = 0.0;

    /* torque authority ramps in; final cap sits inside the 4 N*m envelope */
    double ramp = t / T_AUTH; if (ramp > 1.0) ramp = 1.0; if (ramp < 0.0) ramp = 0.0;
    double umax = U_MIN + (U_MAX - U_MIN) * ramp;
    if (cmd >  umax) cmd =  umax;
    if (cmd < -umax) cmd = -umax;
    if (cmd >  U_MAX) cmd =  U_MAX;
    if (cmd < -U_MAX) cmd = -U_MAX;
    if (!(cmd == cmd)) cmd = 0.0;                 /* non-finite guard */

    /* self slew-limit against own prior output -> cage slew clip never fires */
    double step = cmd - u_out;
    if (step >  SLEW) step =  SLEW;
    if (step < -SLEW) step = -SLEW;
    u_out += step;

    ref_prev  = ref;
    refd_prev = refd;
    return u_out;
}
