{

        return 0.0;

    const double q  = obs->angle_deg;
    const double qd = obs->velocity_dps;

    if (!s_init) {

        s_t0 = obs->t;
        s_q_start = q;

        /* cosine blend peak velocity = (pi/2)*d/T */
        double T = (PI_D * 0.5) * d / APPROACH_VMAX_DPS;
        s_approach_T = T > APPROACH_MIN_S ? T : APPROACH_MIN_S;

            s_cycle_bad = true;

    tau = clampd(tau, s_tau_last - TAU_SLEW, s_tau_last + TAU_SLEW);

    if (!isfinite(tau)) tau = 0.0;
    s_tau_last = tau;
    return tau;
}
