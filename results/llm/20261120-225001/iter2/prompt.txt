#define CT_PERIOD_S   1.5
#define CT_TICK_S     0.002
#define CT_THETA_LO   20.0      /* deg */
#define CT_THETA_HI   85.0      /* deg */

void ctrl_init(void)

{

    *dds = 60.0 * u - 180.0 * u2 + 120.0 * u3;
}

{
    if (x > lim)  return lim;
    if (x < -lim) return -lim;
    return x;
}
