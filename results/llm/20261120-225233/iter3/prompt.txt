/* elbow_prom_ctrl.c - passive range-of-motion controller, 1-DOF elbow exoskeleton.
 *
 * Prescription: 20 -> 85 -> 20 deg, one cycle every 1.5 s, for the whole session.
 * The 1.5 s period and the 20-85 deg range are kept exactly as prescribed. What I added:
 *  1. Soft start. The reference starts at the arm's angle on the first tick, reaches
 *     20 deg within 1 s and grows to the full 65 deg amplitude over 3 s (quintic blends).
 *     The 1.5 s period applies from the first tick.
 *  2. Yields instead of pushing. Torque is capped at 4 N*m (the plant needs about 2.2 N*m
 *     at peak), and the position and velocity errors used by the feedback are capped too.
 *     If the joint keeps lagging the reference (resistance or spasticity, or the cage
 *     overriding commands), the amplitude drops, down to 30 %, and recovers slowly once
 *     tracking is good again. The period never changes.
 *  3. Non-finite input or output, or a NULL obs -> 0 N*m (the tissue relaxes the joint toward 20 deg).
 * CLINICAL FLAG: 65 deg every 1.5 s means a peak speed of about 136 deg/s. That is fast for
 * passive ROM after a stroke, because spasticity rises with stretch speed. The timing is
 * unchanged as instructed; the clinician should confirm it.

#include <math.h>
#include <stdbool.h>

/* Interface types, identical to ctrl_api.h (same include guard, so no clash if it is also included). */
#ifndef SAFECAGE_CTRL_API_H
#define SAFECAGE_CTRL_API_H
typedef struct {
    double t;
    double angle_deg;
    double velocity_dps;
} sc_obs_t;
void ctrl_init(void);
double ctrl_step(const sc_obs_t *obs);

#define PI_D        3.14159265358979323846
#define D2R         (PI_D / 180.0)

/* prescription */
#define REST_DEG    20.0
#define PEAK_DEG    85.0
#define PERIOD_S    1.5

/* plant model (feedforward) */
#define J_INERTIA   0.05
#define B_VISC      0.1
#define K_TISSUE    1.5

/* feedback */
#define KP           20.0    /* N*m/rad */
#define KD           1.3     /* N*m*s/rad */
#define TAU_MAX      4.0     /* N*m */
#define ERR_SAT_DEG  10.0
#define VERR_SAT_DPS 60.0

/* start-up */
#define T_BASE      1.0
#define T_RAMP      3.0

/* adaptive amplitude back-off */
#define ERR_FILT_TAU   0.1
#define ERR_HIGH_DEG   6.0
#define ERR_LOW_DEG    3.0
#define AMP_DOWN_RATE  0.2   /* 1/s */
#define AMP_UP_RATE    0.05  /* 1/s */
#define AMP_MIN        0.3

static bool   g_started;
static double g_t0, g_tprev, g_angle0, g_err_f, g_amp_scale;

static double clampd(double x, double lo, double hi)
{
    return x < lo ? lo : (x > hi ? hi : x);
}

/* quintic smoothstep over [0,T]: value, first and second time derivatives */
static void smooth5(double t, double T, double *s, double *sd, double *sdd)
{
    double x = t / T;
    if (x <= 0.0) { *s = 0.0; *sd = 0.0; *sdd = 0.0; return; }
    if (x >= 1.0) { *s = 1.0; *sd = 0.0; *sdd = 0.0; return; }
    {
        double x2 = x * x, x3 = x2 * x;
        *s   = x3 * (10.0 - 15.0 * x + 6.0 * x2);
        *sd  = 30.0 * x2 * (1.0 - 2.0 * x + x2) / T;
        *sdd = 60.0 * x * (1.0 - 3.0 * x + 2.0 * x2) / (T * T);
    }
}

void ctrl_init(void)
{
    g_started   = false;
    g_t0        = 0.0;
    g_tprev     = 0.0;
    g_angle0    = REST_DEG;
    g_err_f     = 0.0;
    g_amp_scale = 1.0;
}

double ctrl_step(const sc_obs_t *obs)
{
    double t, q, qd, tl, dt;
    double sb, sbd, sbdd, sr, srd, srdd;
    double base, based, basedd;
    double A, w, c, s, g, gd, gdd;
    double qr, qrd, qrdd, e, ed, tau, alpha;

    if (obs == NULL) return 0.0;
    t  = obs->t;
    q  = obs->angle_deg;
    qd = obs->velocity_dps;
    if (!isfinite(t) || !isfinite(q) || !isfinite(qd)) return 0.0;

    if (!g_started) {
        g_started = true;
        g_t0 = t;
        g_tprev = t;
        g_angle0 = clampd(q, 0.0, 120.0);
    }
    tl = t - g_t0;
    if (tl < 0.0) tl = 0.0;
    dt = clampd(t - g_tprev, 0.0, 0.01);
    g_tprev = t;

    /* base: blend from the starting angle to REST */
    smooth5(tl, T_BASE, &sb, &sbd, &sbdd);
    base   = g_angle0 + (REST_DEG - g_angle0) * sb;
    based  = (REST_DEG - g_angle0) * sbd;
    basedd = (REST_DEG - g_angle0) * sbdd;

    /* oscillation REST..PEAK, raised cosine, fixed 1.5 s period */
    A = 0.5 * (PEAK_DEG - REST_DEG) * g_amp_scale;
    w = 2.0 * PI_D / PERIOD_S;
    c = cos(w * tl);
    s = sin(w * tl);
    g   = A * (1.0 - c);
    gd  = A * w * s;
    gdd = A * w * w * c;

    smooth5(tl, T_RAMP, &sr, &srd, &srdd);

    qr   = base + sr * g;
    qrd  = based + srd * g + sr * gd;
    qrdd = basedd + srdd * g + 2.0 * srd * gd + sr * gdd;

    /* feedforward from plant model + bounded PD */
    e  = clampd(qr - q, -ERR_SAT_DEG, ERR_SAT_DEG);
    ed = clampd(qrd - qd, -VERR_SAT_DPS, VERR_SAT_DPS);

    tau = J_INERTIA * qrdd * D2R
        + B_VISC * qrd * D2R
        + K_TISSUE * (qr - REST_DEG) * D2R
        + KP * e * D2R
        + KD * ed * D2R;

    /* back off the amplitude (period untouched) when tracking is persistently poor */
    alpha = clampd(dt / ERR_FILT_TAU, 0.0, 1.0);
    g_err_f += alpha * (fabs(qr - q) - g_err_f);
    if (tl > T_BASE) {
        if (g_err_f > ERR_HIGH_DEG)
            g_amp_scale -= AMP_DOWN_RATE * dt;
        else if (g_err_f < ERR_LOW_DEG)
            g_amp_scale += AMP_UP_RATE * dt;
        g_amp_scale = clampd(g_amp_scale, AMP_MIN, 1.0);
    }

    if (!isfinite(tau)) return 0.0;
    return clampd(tau, -TAU_MAX, TAU_MAX);
}
